
<script>(function(){
  // DIAG: any startup exception becomes visible text (no more silent grey screens)
  window.__NW_ERR=null;

  window.addEventListener("unhandledrejection",function(e){ window.__NW_ERR="rejection: "+(e.reason&&e.reason.message||e.reason); });
  if(!document.getElementById("nwStage")){
    var st=document.createElement("div"); st.id="nwStage";
    st.style.cssText="position:fixed;right:10px;top:10px;z-index:60;font:10px ui-monospace,monospace;color:#8AF2FF;background:rgba(8,16,28,.85);border:1px solid rgba(255,45,120,.8);border-radius:6px;padding:6px 8px;letter-spacing:.06em;white-space:pre";
    st.textContent="NW loader";
    document.body.appendChild(st);
  }
  window.__NW_STAGE="blobs";
  setInterval(function(){
    var el=document.getElementById("nwStage"); if(!el) return;
    var stg=window.__NW_STAGE||"blobs";
    var err=window.__NW_ERR?("\nERR: "+window.__NW_ERR):"";
    var mm=window.__NW_MMERR?("\nMM: "+window.__NW_MMERR):"";
    var menu=window.__NW_MENU?(" | "+window.__NW_MENU):"";
    el.textContent="NW "+stg+menu+err+mm;
    var bad=err||stg==="blobs"||stg==="eval";
    el.style.borderColor=bad?"rgba(255,45,120,.8)":"rgba(0,245,255,.35)";

  try{
    for(var i=0;i<ORDER.length;i++){
      var name=ORDER[i];

      if(typeof src!=="string") throw new Error("module missing in payload: "+name);

               return 'from "'+url[dep]+'"';
             });
      url[name]=URL.createObjectURL(new Blob([src],{type:"text/javascript"}));
    }
  }catch(e){ return fail(e); }
  import(url.main).then(function(){
    var s=document.getElementById("bootStatus"); if(s) s.remove();

  }
